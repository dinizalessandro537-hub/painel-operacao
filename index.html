// ========================================== // API DE CONEXÃO COM O GITHUB PAGES // ========================================== // 1. LER DADOS (Manda os dados da planilha para a tela do GitHub) function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Agora sim, buscando pelo nome EXATO da aba que está na sua foto! var abaDados = ss.getSheetByName("Respostas ao formulário 1") || ss.getSheets()[0]; if (!abaDados) { return ContentService.createTextOutput(JSON.stringify({erro: "Aba não encontrada"})) .setMimeType(ContentService.MimeType.JSON); } var ultimaLinha = abaDados.getLastRow(); var dados = []; // Pega os dados da linha 2 até a última if (ultimaLinha > 1) { var valores = abaDados.getRange(2, 1, ultimaLinha - 1, 5).getDisplayValues(); for (var i = 0; i < valores.length; i++) { dados.push({ data: valores[i][0], codigo: valores[i][1], corredor: valores[i][2], alerta: valores[i][3], dia: valores[i][4] }); } } var resposta = { total: dados.length, registros: dados }; // Libera o acesso para o GitHub ler sem bloqueio return ContentService.createTextOutput(JSON.stringify(resposta)) .setMimeType(ContentService.MimeType.JSON); } // 2. RECEBER DADOS (Pega a melhoria escrita no GitHub e salva na aba MELHORIAS) function doPost(e) { try { var dadosRecebidos = JSON.parse(e.postData.contents); if (dadosRecebidos.tipo === "melhoria") { var ss = SpreadsheetApp.getActiveSpreadsheet(); var abaMelhorias = ss.getSheetByName("MELHORIAS"); if (!abaMelhorias) { abaMelhorias = ss.insertSheet("MELHORIAS"); } if (abaMelhorias.getLastRow() === 0) { abaMelhorias.appendRow(["DATA E HORA", "SUGESTÃO", "STATUS"]); abaMelhorias.getRange("A1:C1").setBackground("#FFE600").setFontWeight("bold"); } var dataAtual = new Date().toLocaleString("pt-BR"); abaMelhorias.appendRow([dataAtual, dadosRecebidos.texto, "PENDENTE"]); return ContentService.createTextOutput(JSON.stringify({sucesso: true})) .setMimeType(ContentService.MimeType.JSON); } } catch (erro) { return ContentService.createTextOutput(JSON.stringify({sucesso: false, erro: erro.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
